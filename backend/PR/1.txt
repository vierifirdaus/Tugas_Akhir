from flask import Flask, request, jsonify
import ast

app = Flask(__name__)

def parse_code(code):
    """
    Function to parse the Python code and extract classes, methods, and functions.
    This will also separate the main body of code outside of classes and functions.
    """
    tree = ast.parse(code)
    result = {'class': [], 'function': [], 'main': ''}

    # Extract functions and classes
    main_code = []
    # print(tree)
    for node in tree.body:
        print("Node:", ast.unparse(node))
        if isinstance(node, ast.FunctionDef):
            func_data = {
                'id': len(result['function']) + 1,
                'functionName': node.name,
                'code': ast.unparse(node)
            }
            result['function'].append(func_data)
        elif isinstance(node, ast.ClassDef):
            class_data = {
                'id': len(result['class']) + 1,
                'classname': node.name,
                'method': []
            }
            # Extract methods within the class
            for class_node in node.body:
                if isinstance(class_node, ast.FunctionDef):
                    method_data = {
                        'methodName': class_node.name,
                        'code': ast.unparse(class_node)
                    }
                    class_data['method'].append(method_data)
            result['class'].append(class_data)
        else:
            # Capture any code outside classes and functions as 'main'
            main_code.append(ast.unparse(node))

    result['main'] = "\n".join(main_code)  # Join all main code parts together

    return result

@app.route('/parse', methods=['POST'])
def parse():
    code = request.json.get('code')
    if not code:
        return jsonify({'error': 'No code provided'}), 400

    parsed_data = parse_code(code)
    return jsonify(parsed_data)

if __name__ == '__main__':
    app.run(debug=True)
